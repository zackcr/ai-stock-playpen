function fmtMoney(n, digits = 2) {
  return n.toLocaleString(undefined, { style: "currency", currency: "USD", minimumFractionDigits: digits, maximumFractionDigits: digits });
}
function fmtPct(n) {
  const v = (n * 100).toFixed(1);
  return `${n >= 0 ? "+" : ""}${v}%`;
}
function fmtMarketCap(n) {
  // Finnhub reports marketCapitalization in millions of USD.
  if (n >= 1e6) return `$${(n / 1e6).toFixed(2)}T`;
  if (n >= 1e3) return `$${(n / 1e3).toFixed(2)}B`;
  return `$${n.toFixed(0)}M`;
}

function dipPct(price, high) {
  return (price - high) / high;
}
function rangePos(price, low, high) {
  if (high === low) return 0;
  return Math.min(1, Math.max(0, (price - low) / (high - low)));
}
function classifyTrend(ret13w, ret26w) {
  if (ret13w == null && ret26w == null) return null;
  const signals = [ret13w, ret26w].filter((r) => r != null);
  const up = signals.filter((r) => r > 0.02).length;
  const down = signals.filter((r) => r < -0.02).length;
  if (up === signals.length && up > 0) return { label: "Pullback in uptrend", color: "#5DCAA5", bg: "#04342C" };
  if (down === signals.length && down > 0) return { label: "Downtrend — caution", color: "#E1636B", bg: "#501313" };
  return { label: "Mixed / choppy", color: "#B4B2A9", bg: "#2C2C2A" };
}
function getBuySignalTickers(stocks, buyTarget) {
  return stocks.filter((s) => !s.archived && dipPct(s.price, s.high) <= buyTarget).map((s) => s.ticker);
}
function aggregateLots(lots) {
  if (!lots || !lots.length) return { shares: null, cost: null };
  const shares = lots.reduce((sum, l) => sum + l.shares, 0);
  if (shares <= 0) return { shares: null, cost: null };
  const totalCost = lots.reduce((sum, l) => sum + l.shares * l.cost, 0);
  return { shares, cost: totalCost / shares };
}
function dateKey(ts) {
  const d = new Date(ts);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
}
function getTrend(points, windowDays) {
  if (!points || points.length < 2) return null;
  const cutoff = Date.now() - windowDays * 86400000;
  const inWindow = points.filter((p) => p.t >= cutoff);
  const series = inWindow.length >= 2 ? inWindow : points.slice(-2);
  const first = series[0];
  const last = series[series.length - 1];
  if (!first || !last || first.price === 0) return null;
  return { pct: (last.price - first.price) / first.price, points: series };
}
function sparklinePath(points, width, height) {
  if (!points || points.length < 2) return "";
  const prices = points.map((p) => p.price);
  const min = Math.min(...prices);
  const max = Math.max(...prices);
  const range = max - min || 1;
  return points
    .map((p, i) => {
      const x = (i / (points.length - 1)) * width;
      const y = height - ((p.price - min) / range) * height;
      return `${i === 0 ? "M" : "L"}${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(" ");
}
function industryToCategory(industry) {
  if (!industry) return null;
  const lower = industry.toLowerCase();
  const rules = [
    [/semiconductor/, "Chipmaker"],
    [/metal|mining/, "Material Input"],
    [/data storage|storage/, "Memory"],
    [/electrical equip|electronic equip|communication equip|industrial/, "Infrastructure"],
    [/software|it services|internet|cloud/, "IaaS"],
    [/exchange traded|etf/, "ETF"],
  ];
  for (const [re, category] of rules) if (re.test(lower)) return category;
  return industry;
}

function Modal({ titleId, title, subtitle, onClose, width = 360, children }) {
  const containerRef = useRef(null);
  const previouslyFocused = useRef(null);
  const onCloseRef = useRef(onClose);
  onCloseRef.current = onClose;

  useEffect(() => {
    previouslyFocused.current = document.activeElement;
    const focusables = () =>
      containerRef.current
        ? Array.prototype.slice
            .call(containerRef.current.querySelectorAll('button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'))
            .filter((el) => !el.disabled && el.offsetParent !== null)
        : [];
    const first = focusables()[0];
    (first || containerRef.current) && (first || containerRef.current).focus();

    function handleKey(e) {
      if (e.key === "Escape") {
        e.stopPropagation();
        onCloseRef.current();
        return;
      }
      if (e.key === "Tab") {
        const list = focusables();
        if (!list.length) return;
        const firstEl = list[0];
        const lastEl = list[list.length - 1];
        if (e.shiftKey && document.activeElement === firstEl) {
          e.preventDefault();
          lastEl.focus();
        } else if (!e.shiftKey && document.activeElement === lastEl) {
          e.preventDefault();
          firstEl.focus();
        }
      }
    }
    document.addEventListener("keydown", handleKey, true);
    return () => {
      document.removeEventListener("keydown", handleKey, true);
      if (previouslyFocused.current && previouslyFocused.current.focus) previouslyFocused.current.focus();
    };
  }, []);

  return (
    <div
      style={{ position: "fixed", inset: 0, background: "rgba(0,0,0,0.55)", display: "flex", alignItems: "center", justifyContent: "center", zIndex: 50, padding: 20, overflowY: "auto" }}
      onClick={onClose}
    >
      <div
        ref={containerRef}
        role="dialog"
        aria-modal="true"
        aria-labelledby={titleId}
        tabIndex={-1}
        style={{ background: "#131C2E", border: "1px solid #223049", borderRadius: 12, padding: 22, width, maxHeight: "85vh", overflowY: "auto", outline: "none" }}
        onClick={(e) => e.stopPropagation()}
      >
        <h2 id={titleId} style={{ color: "#E8EDF5", fontSize: 16, fontWeight: 700, marginBottom: subtitle ? 4 : 16, marginTop: 0 }}>
          {title}
        </h2>
        {subtitle && <div style={{ color: "#7C89A3", fontSize: 12, marginBottom: 16 }}>{subtitle}</div>}
        {children}
      </div>
    </div>
  );
}

      ]);
    } catch (e) {
      throw new Error(`${ticker}: network/CORS error — ${e.message || "request never completed"}`);
    }

    if (!qRes.ok) {
      let body = "";
      try {
        body = (await qRes.text()).slice(0, 120);
      } catch (e2) {
        // ignore
      }
      throw new Error(`${ticker}: quote HTTP ${qRes.status} ${qRes.statusText}${body ? " — " + body : ""}`);
    }
    const q = await qRes.json();
    const price = q && typeof q.c === "number" && q.c > 0 ? q.c : null;
    if (price == null) throw new Error(`${ticker}: quote returned no price (${JSON.stringify(q)})`);
    const prevClose = typeof q.pc === "number" && q.pc > 0 ? q.pc : null;
    const dayOpen = typeof q.o === "number" && q.o > 0 ? q.o : null;
    const dayHigh = typeof q.h === "number" && q.h > 0 ? q.h : null;
    const dayLow = typeof q.l === "number" && q.l > 0 ? q.l : null;

    // Metric (52w high/low, 13w/26w return, and the extra fundamentals) is treated as
    // best-effort — a failure here shouldn't block the price update, since some free-tier
    // keys restrict parts of it.
    let high = null, low = null, ret13w = null, ret26w = null;
    let marketCap = null, peRatio = null, dividendYield = null, beta = null, metricError = null;
    if (mRes && mRes.ok) {
      try {
        const m = await mRes.json();
        const metric = (m && m.metric) || {};
        high = typeof metric["52WeekHigh"] === "number" ? metric["52WeekHigh"] : null;
        low = typeof metric["52WeekLow"] === "number" ? metric["52WeekLow"] : null;
        ret13w = typeof metric["13WeekPriceReturnDaily"] === "number" ? metric["13WeekPriceReturnDaily"] / 100 : null;
        ret26w = typeof metric["26WeekPriceReturnDaily"] === "number" ? metric["26WeekPriceReturnDaily"] / 100 : null;
        marketCap = typeof metric.marketCapitalization === "number" ? metric.marketCapitalization : null;
        peRatio = typeof metric.peBasicExclExtraTTM === "number" ? metric.peBasicExclExtraTTM : null;
        dividendYield = typeof metric.dividendYieldIndicatedAnnual === "number" ? metric.dividendYieldIndicatedAnnual / 100 : null;
        beta = typeof metric.beta === "number" ? metric.beta : null;
      } catch (e3) {
        metricError = "metric parse error";
      }
    } else if (mRes) {
      metricError = `metric HTTP ${mRes.status}`;
    }
    return { price, high, low, ret13w, ret26w, prevClose, dayOpen, dayHigh, dayLow, marketCap, peRatio, dividendYield, beta, metricError };
  }

  async function refreshLive() {
    if (!settings.apiKey) {
      openSettings();
      return;
    }
    setRefreshing(true);
    setRefreshErrors([]);
    const errors = [];
    const results = {};
    const active = stocks.filter((s) => !s.archived);
    for (let i = 0; i < active.length; i++) {
      const t = active[i].ticker;
      try {
        results[t] = await fetchQuote(t, settings.apiKey);
      } catch (e) {
        errors.push(e && e.message ? e.message : `${t}: unknown error`);
      }
      await new Promise((r) => setTimeout(r, 120));
    }
    const next = stocks.map((s) => {
      const r = results[s.ticker];
      if (!r) return s;
      const high = r.high != null ? Math.max(r.high, r.price) : Math.max(s.high, r.price);
      const low = r.low != null ? Math.min(r.low, r.price) : Math.min(s.low, r.price);
      return {
        ...s,
        price: r.price,
        high,
        low,
        ret13w: r.ret13w != null ? r.ret13w : s.ret13w != null ? s.ret13w : null,
        ret26w: r.ret26w != null ? r.ret26w : s.ret26w != null ? s.ret26w : null,
        prevClose: r.prevClose != null ? r.prevClose : s.prevClose != null ? s.prevClose : null,
        dayOpen: r.dayOpen != null ? r.dayOpen : s.dayOpen != null ? s.dayOpen : null,
        dayHigh: r.dayHigh != null ? r.dayHigh : s.dayHigh != null ? s.dayHigh : null,
        dayLow: r.dayLow != null ? r.dayLow : s.dayLow != null ? s.dayLow : null,
        marketCap: r.marketCap != null ? r.marketCap : s.marketCap != null ? s.marketCap : null,
        peRatio: r.peRatio != null ? r.peRatio : s.peRatio != null ? s.peRatio : null,
        dividendYield: r.dividendYield != null ? r.dividendYield : s.dividendYield != null ? s.dividendYield : null,
        beta: r.beta != null ? r.beta : s.beta != null ? s.beta : null,
      };
    });
    setStocks(next);
    persist(next);
    const priceMap = {};
    Object.keys(results).forEach((t) => (priceMap[t] = results[t].price));
    if (Object.keys(priceMap).length) recordSnapshot(priceMap);

    const currentSignals = getBuySignalTickers(next, BUY_TARGET);
    const freshSignals = currentSignals.filter((t) => !(lastSeenBuySignals || []).includes(t));
    if (freshSignals.length) {
      setNewSignals(freshSignals);
      if (settings.notifyEnabled && typeof Notification < "u" && Notification.permission === "granted") {
        try {
          new Notification("New buy-zone signal", {
            body: freshSignals.length === 1
              ? `${freshSignals[0]} just hit your dip target.`
              : `${freshSignals.length} tickers just hit your dip target: ${freshSignals.join(", ")}`,
          });
        } catch (e) {
          // notification blocked or unsupported mid-session — non-fatal
        }
      }
    }
    setLastSeenBuySignals(currentSignals);
    try {
      await window.storage.set(LASTSEEN_KEY, JSON.stringify(currentSignals), false);
    } catch (e) {
      // ignore
    }

    setRefreshErrors(errors);
    setLastUpdated(new Date());
    setRefreshing(false);
    if (settings.sheetSync.enabled && settings.sheetSync.autoPush) {
      pushToSheet(next);
    }
  }

  async function pushToSheet(stocksToSend) {
    if (!settings.sheetSync.url) return;
    setPushing(true);
    setPushMsg("");
    const payload = {
      updated: new Date().toISOString(),
      stocks: (stocksToSend || stocks)
        .filter((s) => !s.archived)
        .map((s) => {
          const dip = dipPct(s.price, s.high);
          const aboveLow = s.low > 0 ? (s.price - s.low) / s.low : 0;
          const rp = rangePos(s.price, s.low, s.high);
          const value = s.shares ? s.shares * s.price : null;
          const cost = s.shares ? s.shares * s.cost : null;
          return {
            ticker: s.ticker,
            name: s.name,
            category: s.category,
            price: s.price,
            high: s.high,
            low: s.low,
            dip,
            aboveLow,
            buyTarget: BUY_TARGET,
            rangePosition: rp,
            shares: s.shares,
            cost: s.cost,
            value,
            returnPct: cost ? (value - cost) / cost : null,
            returnDollar: cost ? value - cost : null,
            updated: new Date().toISOString(),
          };
        }),
    };
    try {
      await fetch(settings.sheetSync.url, {
        method: "POST",
        mode: "no-cors",
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        body: JSON.stringify(payload),
      });
      setPushMsg("Pushed — check your sheet to confirm.");
    } catch (e) {
      setPushMsg("Push failed. Check the web app URL in settings.");
    } finally {
      setPushing(false);
      setTimeout(() => setPushMsg(""), 4000);
    }
  }

  function downloadFile(content, filename, mimeType) {
    const blob = new Blob([content], { type: mimeType });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }

  function importBackup(file) {
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const parsed = JSON.parse(reader.result);
        if (!parsed || !Array.isArray(parsed.stocks)) throw new Error("missing stocks array");
        const nextStocks = parsed.stocks.map((s) => ({
          archived: false,
          ret13w: null,
          ret26w: null,
          prevClose: null,
          dayOpen: null,
          dayHigh: null,
          dayLow: null,
          marketCap: null,
          peRatio: null,
          dividendYield: null,
          beta: null,
          ...s,
        }));
        setStocks(nextStocks);
        persist(nextStocks);
        if (parsed.settings && typeof parsed.settings === "object") {
          const nextSettings = {
            ...DEFAULT_SETTINGS,
            ...parsed.settings,
            sheetSync: { ...DEFAULT_SETTINGS.sheetSync, ...(parsed.settings.sheetSync || {}) },
          };
          setSettings(nextSettings);
          persistSettings(nextSettings);
        }
        if (parsed.history && typeof parsed.history === "object") {
          setHistory(parsed.history);
          persistHistory(parsed.history);
        }
        setImportMsg(`Imported ${nextStocks.length} tickers.`);
      } catch (e) {
        setImportMsg("Import failed — file isn't a valid backup.");
      }
      setTimeout(() => setImportMsg(""), 4000);
    };
    reader.readAsText(file);
  }

  function exportCsv() {
    const headers = [
      "Ticker", "Company Name", "Category", "Current Price", "52-Week High", "52-Week Low",
      "Dip %", "Shares", "Cost Basis", "Current Value", "Total Return %", "Archived",
    ];
    const rows = stocks.map((s) => {
      const dip = dipPct(s.price, s.high);
      const value = s.shares ? s.shares * s.price : "";
      const cost = s.shares ? s.shares * s.cost : "";
      const returnPct = s.shares && cost ? (value - cost) / cost : "";
      return [
        s.ticker, s.name, s.category, s.price, s.high, s.low,
        (dip * 100).toFixed(1), s.shares || "", s.cost || "", value,
        returnPct === "" ? "" : (returnPct * 100).toFixed(1), s.archived ? "yes" : "no",
      ];
    });
    const csv = [headers, ...rows]
      .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(","))
      .join("\n");
    downloadFile(csv, `ai-stock-playpen-${dateKey(Date.now())}.csv`, "text/csv");
  }

  async function submitAdd() {
    const ticker = addDraft.ticker.trim().toUpperCase();
    const name = addDraft.name.trim();
    const category = addDraft.category.trim() || "Uncategorized";
    if (!ticker) return setAddError("Enter a ticker symbol.");
    if (stocks.some((s) => s.ticker === ticker)) return setAddError(`${ticker} is already on the list.`);

    let price = parseFloat(addDraft.price);
    let high = parseFloat(addDraft.high);
    let low = parseFloat(addDraft.low);

    if ((isNaN(price) || isNaN(high) || isNaN(low)) && settings.apiKey) {
      try {
        const live = await fetchQuote(ticker, settings.apiKey);
        price = live.price;
        high = live.high != null ? live.high : live.price;
        low = live.low != null ? live.low : live.price;
      } catch (e) {
        return setAddError(`Couldn't fetch live data for ${ticker}. Enter price/high/low manually.`);
      }
    }
    if (isNaN(price) || price <= 0) return setAddError("Enter a valid current price, or connect live data.");
    if (isNaN(high)) high = price;
    if (isNaN(low)) low = price;
    if (low > high) [low, high] = [high, low];

    const next = [
      ...stocks,
      {
        ticker, name: name || ticker, category, price, high, low, shares: null, cost: null, archived: false,
        ret13w: null, ret26w: null, prevClose: null, dayOpen: null, dayHigh: null, dayLow: null,
        marketCap: null, peRatio: null, dividendYield: null, beta: null, lots: [],
      },
    ];
    setStocks(next);
    persist(next);
    recordSnapshot({ [ticker]: price });
    setShowAdd(false);
  }

  async function fetchInsights(ticker) {
    if (!settings.apiKey) return;
    setInsights((prev) => ({ ...prev, [ticker]: { ...(prev[ticker] || {}), loading: true } }));
    const apiKey = settings.apiKey;
    const today = new Date();
    const newsFrom = new Date(today.getTime() - 14 * 86400000).toISOString().slice(0, 10);
    const newsTo = today.toISOString().slice(0, 10);
    const earningsFrom = today.toISOString().slice(0, 10);
    const earningsTo = new Date(today.getTime() + 120 * 86400000).toISOString().slice(0, 10);

    const [newsResult, earningsResult, recResult] = await Promise.allSettled([
      fetch(`https://finnhub.io/api/v1/company-news?symbol=${encodeURIComponent(ticker)}&from=${newsFrom}&to=${newsTo}&token=${apiKey}`).then((r) =>
        r.ok ? r.json() : Promise.reject(new Error(`news HTTP ${r.status}`))
      ),
      fetch(`https://finnhub.io/api/v1/calendar/earnings?from=${earningsFrom}&to=${earningsTo}&symbol=${encodeURIComponent(ticker)}&token=${apiKey}`).then((r) =>
        r.ok ? r.json() : Promise.reject(new Error(`earnings HTTP ${r.status}`))
      ),
      fetch(`https://finnhub.io/api/v1/stock/recommendation?symbol=${encodeURIComponent(ticker)}&token=${apiKey}`).then((r) =>
        r.ok ? r.json() : Promise.reject(new Error(`recommendation HTTP ${r.status}`))
      ),
    ]);

    const news = newsResult.status === "fulfilled" && Array.isArray(newsResult.value) ? newsResult.value.slice(0, 3) : null;
    const newsError = newsResult.status === "rejected" ? newsResult.reason.message : null;

    let nextEarnings = null, earningsError = null;
    if (earningsResult.status === "fulfilled") {
      const calendar = (earningsResult.value && earningsResult.value.earningsCalendar) || [];
      if (calendar.length) nextEarnings = calendar.slice().sort((a, b) => (a.date < b.date ? -1 : 1))[0].date;
    } else {
      earningsError = earningsResult.reason.message;
    }

    let recommendation = null, recError = null;
    if (recResult.status === "fulfilled" && Array.isArray(recResult.value) && recResult.value.length) {
      recommendation = recResult.value[0];
    } else if (recResult.status === "rejected") {
      recError = recResult.reason.message;
    }

    setInsights((prev) => ({
      ...prev,
      [ticker]: { news, newsError, nextEarnings, earningsError, recommendation, recError, loading: false, fetchedAt: Date.now() },
    }));
  }

  function convertToLots() {
    const stock = stocks.find((s) => s.ticker === editing);
    if (!stock) return;
    const shares = parseFloat(draft.shares);
    const cost = parseFloat(draft.cost);
    const seedLots =
      !isNaN(shares) && shares > 0 && !isNaN(cost) && cost > 0
        ? [{ id: `${stock.ticker}-legacy`, shares, cost, date: dateKey(Date.now()) }]
        : [];
    const next = stocks.map((s) => {
      if (s.ticker !== editing) return s;
      const agg = aggregateLots(seedLots);
      return { ...s, lots: seedLots, shares: agg.shares, cost: agg.cost };
    });
    setStocks(next);
    persist(next);
  }

  function addLot() {
    const shares = parseFloat(lotDraft.shares);
    const cost = parseFloat(lotDraft.cost);
    if (isNaN(shares) || shares <= 0 || isNaN(cost) || cost <= 0) return;
    const next = stocks.map((s) => {
      if (s.ticker !== editing) return s;
      const existingLots =
        s.lots && s.lots.length
          ? s.lots
          : s.shares
            ? [{ id: `${s.ticker}-legacy`, shares: s.shares, cost: s.cost, date: dateKey(Date.now()) }]
            : [];
      const lots = [...existingLots, { id: `${s.ticker}-${Date.now()}`, shares, cost, date: dateKey(Date.now()) }];
      const agg = aggregateLots(lots);
      return { ...s, lots, shares: agg.shares, cost: agg.cost };
    });
    setStocks(next);
    persist(next);
    setLotDraft({ shares: "", cost: "" });
  }

  function sellLot(ticker, lotId) {
    const stock = stocks.find((s) => s.ticker === ticker);
    if (!stock || !stock.lots) return;
    const lot = stock.lots.find((l) => l.id === lotId);
    if (!lot) return;
    const gain = lot.shares * (stock.price - lot.cost);
    const nextRealized = [
      ...realizedGains,
      { ticker, shares: lot.shares, cost: lot.cost, sellPrice: stock.price, date: new Date().toISOString(), gain },
    ];
    setRealizedGains(nextRealized);
    persistRealized(nextRealized);
    const next = stocks.map((s) => {
      if (s.ticker !== ticker) return s;
      const lots = s.lots.filter((l) => l.id !== lotId);
      const agg = aggregateLots(lots);
      return { ...s, lots, shares: agg.shares, cost: agg.cost };
    });
    setStocks(next);
    persist(next);
  }

  const categories = ["All", ...Array.from(new Set(stocks.filter((s) => !s.archived).map((s) => s.category)))];
  const archivedCount = stocks.filter((s) => s.archived).length;

  let totalCost = 0, totalValue = 0, buySignals = 0, positionCount = 0;
  const buySignalsByCategory = {};
  stocks.forEach((s) => {
    if (s.shares) {
      totalCost += s.shares * s.cost;
      totalValue += s.shares * s.price;
      positionCount += 1;
    }
    if (!s.archived && dipPct(s.price, s.high) <= BUY_TARGET) {
      buySignals += 1;
      buySignalsByCategory[s.category] = (buySignalsByCategory[s.category] || 0) + 1;
    }
  });
  const totalReturn = totalValue - totalCost;
  const totalReturnPct = totalCost > 0 ? totalReturn / totalCost : 0;
  const portfolioTotal = totalValue + settings.cash;
  const realizedTotal = realizedGains.reduce((sum, r) => sum + r.gain, 0);

  let concentration = null;
  if (buySignals >= 2) {
    const [topCategory, topCount] = Object.entries(buySignalsByCategory).sort((a, b) => b[1] - a[1])[0];
    const share = topCount / buySignals;
    if (share >= 0.5) concentration = { category: topCategory, count: topCount, total: buySignals, share };
  }

  const overweightPositions =
    portfolioTotal > 0
      ? stocks
          .filter((s) => s.shares)
          .map((s) => ({ ticker: s.ticker, share: (s.shares * s.price) / portfolioTotal }))
          .filter((s) => s.share > settings.maxPositionPct)
          .sort((a, b) => b.share - a.share)
      : [];

  const visible = stocks
    .filter((s) => (showArchived ? s.archived : !s.archived))
    .filter((s) => filter === "All" || s.category === filter)
    .filter((s) => !buyOnly || dipPct(s.price, s.high) <= BUY_TARGET)
    .filter((s) => !holdingOnly || !!s.shares)
    .sort((a, b) => dipPct(a.price, a.high) - dipPct(b.price, b.high));

  if (!loaded) {
    return (
      <div style={{ minHeight: 300, display: "flex", alignItems: "center", justifyContent: "center", fontFamily: "ui-monospace, monospace", color: "#8A96AC" }}>
        loading playpen...
      </div>
    );
  }

  return (
    <div style={{ background: "#0B1220", minHeight: "100vh", padding: "32px 20px", fontFamily: "'Inter', ui-sans-serif, system-ui, sans-serif" }}>
      <style>{`
        * { box-sizing: border-box; }
        .mono { font-family: ui-monospace, 'SF Mono', 'Cascadia Code', Consolas, monospace; }
        .pill { padding: 4px 10px; border-radius: 999px; font-size: 11px; font-weight: 600; letter-spacing: 0.03em; text-transform: uppercase; }

            </AboutSection>

            <AboutSection title="Range position">
              Where the current price sits between the low (0%) and high (100%) of the 52-week range. This drives the gauge bar's fill.
            </AboutSection>

            <AboutSection title="Buy target / buy zone">
              A dip % threshold you set (default -15%). When a stock's dip % is at or beyond that line, it's flagged "Buy zone" and counted in the buy-signals stat. It's purely a rule on the gauge — it doesn't check volume, trend, or fundamentals, so it's best treated as a screener, not a signal to act on by itself (see the note on the main chat about that).
            </AboutSection>

            <AboutSection title="Trend & sparkline">
              History for the sparkline builds up automatically two ways: once per day whenever you open the app (using whatever price is already stored — no network needed), and whenever you refresh live prices or manually edit one. The sparkline and "±X% / Nd" figure use whichever points fall inside your chosen trend window (7/30/90 days in Settings).
            </AboutSection>

            <AboutSection title="Trend context">
              A colored pill built from Finnhub's 13-week and 26-week price return figures — both positive → "Pullback in uptrend," both negative → "Downtrend — caution," mixed → "Mixed / choppy." This is a proxy for a true 50/200-day moving-average crossover, not the real thing: Finnhub's historical daily candles are paywalled on the free tier, and a separate free source (Stooq) turned out to be unreliable through any public CORS proxy, so a true SMA isn't currently reachable without a paid data source. The 13w/26w figures are the closest free substitute for "is this dip inside a longer uptrend or a longer downtrend."
            </AboutSection>

            <AboutSection title="Concentration warning">
              If half or more of your active buy signals land in a single category, a banner flags it. The dip gauge treats every ticker independently, but AI infrastructure names are correlated — five buy signals in "Memory" on the same day is usually one macro or sector move, not five separate opportunities.
            </AboutSection>

            <AboutSection title="Expanded details">
              Tap "Details" on any card for a bigger price chart (7/30/90-day toggle, built from the same recorded snapshots as the sparkline) plus a stats grid: previous close, day's open, day range, 52-week range, market cap, P/E (TTM), dividend yield, and beta. The extra stats come from the same Finnhub call already used for 52-week high/low, so they cost nothing extra — but they only populate after a live refresh; a fresh add shows placeholders until then. Expanding also lazy-loads recent news, the next earnings date, and analyst buy/hold/sell counts — these use separate Finnhub calls fetched only for the ticker you actually open (not during bulk refresh, to avoid burning through rate limits), cached for a day per ticker.
            </AboutSection>

            <AboutSection title="Cash & position size">
              Set uninvested cash in Settings so "Portfolio total" reflects real dry powder, not just what's already deployed. A warning banner fires if any single position exceeds your configured max position size (default 25% of portfolio total) — sizing risk that dip % alone never surfaces.
            </AboutSection>

            <AboutSection title="What's new & notifications">

If you enable browser notifications in Settings, a "Refresh live" that turns up a new signal will also fire a notification — this only works while the app is open and you trigger a refresh, since true background push would need a server this app doesn't have.
            </AboutSection>

            <AboutSection title="Backup & export">

            </AboutSection>

            <AboutSection title="Position & return">
              Shares × cost basis vs. shares × current price gives return $ and return %. For a simple position, cost basis is one average cost per share, entered manually. If you'd rather track individual buys separately — different prices, different dates — click "Track as multiple lots instead" in the edit panel. Once lots exist, the plain shares/cost fields disappear and become derived automatically (a weighted average across all lots); each lot shows its own paper gain and can be sold independently at the current price, which logs a realized gain/loss entry and removes just that lot. A "Realized P&L" stat card appears once you've sold at least one lot.
            </AboutSection>

            <AboutSection title="Live data">

When adding a new ticker with live data connected, typing a symbol looks up its company name and industry from Finnhub and suggests a category — matched against your existing categories where the industry clearly fits (e.g. "Semiconductors" → Chipmaker), or Finnhub's raw industry label otherwise. Either field can be overwritten by typing in it directly, which stops any further auto-fill for that field on that ticker.
